# Cron Jobs for Adarsh Admin
# ===========================
#
# Edit crontab with: crontab -e
# Then add these entries.
#
# To install all at once:
#   crontab deployment/cron_cleanup.txt
# Or append:
#   cat deployment/cron_cleanup.txt >> /var/spool/cron/crontabs/$(whoami)

# Environment setup
SHELL=/bin/bash
PATH=/usr/local/bin:/usr/bin:/bin
DJANGO_SETTINGS_MODULE=config.settings

# =============================================================================
# Background Task Cleanup
# =============================================================================

# Clean up stale tasks, old results, and temp files
# Runs daily at 2:00 AM
0 2 * * * cd /path/to/project && /path/to/venv/bin/python manage.py cleanup_tasks --all >> /var/log/adarsh/cleanup.log 2>&1

# =============================================================================
# Database Maintenance (if using SQLite)
# =============================================================================

# Vacuum SQLite database weekly (improves performance)
# Runs Sunday at 3:00 AM
# 0 3 * * 0 cd /path/to/project && sqlite3 db.sqlite3 "VACUUM;" >> /var/log/adarsh/db_maintenance.log 2>&1

# =============================================================================
# Log Rotation
# =============================================================================

# Compress old log files daily
# 0 4 * * * find /var/log/adarsh -name "*.log" -mtime +7 -exec gzip {} \;

# Delete very old compressed logs monthly (keep last 30 days)
# 0 5 1 * * find /var/log/adarsh -name "*.log.gz" -mtime +30 -delete

# =============================================================================
# Health Checks (optional)
# =============================================================================

# Ping health endpoint every 5 minutes (for monitoring)
# */5 * * * * curl -s http://localhost/api/health/ > /dev/null || echo "Health check failed at $(date)" >> /var/log/adarsh/health.log

# =============================================================================
# Server Restart (if needed for memory reclamation)
# =============================================================================

# Restart Gunicorn weekly to reclaim any leaked memory
# Runs Sunday at 4:00 AM
# 0 4 * * 0 sudo systemctl restart gunicorn >> /var/log/adarsh/restart.log 2>&1
